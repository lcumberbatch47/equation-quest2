    // Spinner Canvas Physics Drawing

    function drawSpinner(angle = 0) {
      const slices = [1, 2, 3, 4];
      const colors = ['#ef4444', '#3b82f6', '#10b981', '#f59e0b'];
      const sliceAngle = (2 * Math.PI) / 4;

      ctx.clearRect(0, 0, 200, 200);

      for (let i = 0; i < 4; i++) {
        ctx.beginPath();
        ctx.fillStyle = colors[i];
        ctx.moveTo(100, 100);
        ctx.arc(100, 100, 90, angle + i * sliceAngle, angle + (i + 1) * sliceAngle);
        ctx.fill();
        ctx.lineWidth = 3;
        ctx.strokeStyle = '#0f172a';
        ctx.stroke();

        // Numbers
        ctx.save();
        ctx.translate(100, 100);
        ctx.rotate(angle + i * sliceAngle + sliceAngle / 2);
        ctx.fillStyle = '#ffffff';
        ctx.font = 'bold 26px Fredoka';
        ctx.fillText(slices[i], 50, 8);
        ctx.restore();
      }

      // Center Hub
      ctx.beginPath();
      ctx.fillStyle = '#0f172a';
      ctx.arc(100, 100, 20, 0, 2 * Math.PI);
      ctx.fill();
      ctx.lineWidth = 2;
      ctx.strokeStyle = '#334155';
      ctx.stroke();
    }

    function spinWheel() {
      if (isSpinning) return;
      isSpinning = true;
      document.getElementById('spinBtn').disabled = true;

      let spins = 25 + Math.floor(Math.random() * 12);
      let speed = 0.25;
      let count = 0;

      function animate() {
        currentAngle += speed;
        drawSpinner(currentAngle);
        count++;

        if (count < spins) {
          requestAnimationFrame(animate);
        } else {
          isSpinning = false;
          currentSpin = Math.floor(Math.random() * 4) + 1;
          document.getElementById('spinResultText').innerText = `Spun a ${currentSpin}! Answer the equation to move forward.`;
          showQuestionModal();
        }
      }
      animate();
    }

      const diffBadge = document.getElementById('modalDiffBadge');
      if (activeQuestion.diff === 'EASY') {
        diffBadge.className = 'text-xs font-extrabold bg-emerald-500/20 text-emerald-400 border border-emerald-500/40 px-2.5 py-1 rounded-lg';
      } else if (activeQuestion.diff === 'MEDIUM') {
        diffBadge.className = 'text-xs font-extrabold bg-amber-500/20 text-amber-400 border border-amber-500/40 px-2.5 py-1 rounded-lg';
      } else {
        diffBadge.className = 'text-xs font-extrabold bg-purple-500/20 text-purple-400 border border-purple-500/40 px-2.5 py-1 rounded-lg';
      }
      diffBadge.innerText = activeQuestion.diff;

      // Reset Modal UI State
      document.getElementById('feedbackArea').classList.add('hidden');
      document.getElementById('answerContainer').classList.remove('hidden');
      const input = document.getElementById('userAnswerInput');
      input.value = '';
      
      document.getElementById('questionModal').classList.remove('hidden');
      setTimeout(() => input.focus(), 100);
    }

    function parseMathInput(str) {
      if (!str) return null;
      str = str.trim();
      if (str.includes('/')) {
        const parts = str.split('/');
        if (parts.length === 2) {
          const num = parseFloat(parts[0]);
          const den = parseFloat(parts[1]);
          if (!isNaN(num) && !isNaN(den) && den !== 0) return num / den;
        }
      }
      const val = parseFloat(str);
      return isNaN(val) ? null : val;
    }

    function submitAnswer() {
      const inputVal = document.getElementById('userAnswerInput').value;
      const userAns = parseMathInput(inputVal);
      const feedbackArea = document.getElementById('feedbackArea');
      const feedbackHeader = document.getElementById('feedbackHeader');
      const feedbackText = document.getElementById('feedbackText');

      document.getElementById('answerContainer').classList.add('hidden');
      feedbackArea.classList.remove('hidden');

      const animEl = currentPlayer === 1 ? document.getElementById('p1ScoreAnim') : document.getElementById('p2ScoreAnim');

      if (userAns !== null && Math.abs(userAns - activeQuestion.a) < 0.05) {
        // CORRECT ANSWER
        feedbackHeader.className = 'font-bold text-base text-emerald-400 flex items-center gap-2';
        feedbackHeader.innerHTML = `<i class="fa-solid fa-circle-check"></i> CORRECT! (+${activeQuestion.pts} PTS & Move ${currentSpin} Spaces)`;
        feedbackText.innerText = `Great work! ${activeQuestion.steps}`;
        
        // Award points & Move
        if (currentPlayer === 1) {
          p1Pts += activeQuestion.pts;
          p1Pos = Math.min(boardSize - 1, p1Pos + currentSpin);
          if (bonusTiles.includes(p1Pos)) p1Pos = Math.min(boardSize - 1, p1Pos + 1);
        } else {
          p2Pts += activeQuestion.pts;
          p2Pos = Math.min(boardSize - 1, p2Pos + currentSpin);
          if (bonusTiles.includes(p2Pos)) p2Pos = Math.min(boardSize - 1, p2Pos + 1);
        }

        // Trigger Score Overlay Animation
        if (animEl) {
          animEl.innerText = `+${activeQuestion.pts} PTS!`;
          animEl.classList.remove('opacity-0');
          setTimeout(() => animEl.classList.add('opacity-0'), 1800);
        }
      } else {
        // INCORRECT ANSWER (Move forward, 0 points)
        feedbackHeader.className = 'font-bold text-base text-amber-400 flex items-center gap-2';
        feedbackHeader.innerHTML = `<i class="fa-solid fa-circle-xmark"></i> INCORRECT (Moved ${currentSpin} Spaces, 0 PTS)`;
        feedbackText.innerText = `The correct answer was ${activeQuestion.a}. Solution: ${activeQuestion.steps}`;

        // Move player forward, but do NOT award points
        if (currentPlayer === 1) {
          p1Pos = Math.min(boardSize - 1, p1Pos + currentSpin);
          if (bonusTiles.includes(p1Pos)) p1Pos = Math.min(boardSize - 1, p1Pos + 1);
        } else {
          p2Pos = Math.min(boardSize - 1, p2Pos + currentSpin);
          if (bonusTiles.includes(p2Pos)) p2Pos = Math.min(boardSize - 1, p2Pos + 1);
        }

        if (animEl) {
          animEl.innerText = `+0 PTS`;
          animEl.classList.remove('opacity-0');
          setTimeout(() => animEl.classList.add('opacity-0'), 1800);
        }
      }

      renderTokens();

      // Check Victory Condition
      if (p1Pos >= boardSize - 1 || p2Pos >= boardSize - 1) {
        confetti({ particleCount: 150, spread: 80, origin: { y: 0.6 } });
      }
    }

    function closeQuestionModal() {
      document.getElementById('questionModal').classList.add('hidden');

      if (p1Pos >= boardSize - 1 || p2Pos >= boardSize - 1) {
        const winner = p1Pos >= boardSize - 1 ? 'Player 1' : 'Player 2';
        alert(`🎉 CONGRATULATIONS! ${winner} reached the finish line!\nFinal Scores:\nPlayer 1: ${p1Pts} PTS\nPlayer 2: ${p2Pts} PTS`);
        return;
      }

      // Switch Turn
      currentPlayer = currentPlayer === 1 ? 2 : 1;
      const turnBadge = document.getElementById('turnBadge');
      const turnText = document.getElementById('turnText');
      const turnDot = document.getElementById('turnDot');
      const p1Card = document.getElementById('p1Card');
      const p2Card = document.getElementById('p2Card');

      if (currentPlayer === 1) {
        turnBadge.className = 'flex items-center gap-2 px-4 py-2 rounded-xl font-bold text-sm bg-red-500/20 text-red-400 border border-red-500/40';
        turnDot.className = 'w-3 h-3 rounded-full bg-red-500 animate-ping';
        turnText.innerText = 'Player 1 Turn';
        p1Card.classList.add('active-turn-card', 'border-red-500/40');
        p2Card.classList.remove('active-turn-card', 'border-cyan-500/40');
      } else {
        turnBadge.className = 'flex items-center gap-2 px-4 py-2 rounded-xl font-bold text-sm bg-cyan-500/20 text-cyan-400 border border-cyan-500/40';
        turnDot.className = 'w-3 h-3 rounded-full bg-cyan-400 animate-ping';
        turnText.innerText = 'Player 2 Turn';
        p2Card.classList.add('active-turn-card', 'border-cyan-500/40');
        p1Card.classList.remove('active-turn-card', 'border-red-500/40');
      }

      document.getElementById('spinBtn').disabled = false;
      document.getElementById('spinResultText').innerText = 'Click spin to reveal movement potential!';
    }

    // Allow Enter key to submit answer
    document.getElementById('userAnswerInput').addEventListener('keyup', function(event) {
      if (event.key === 'Enter') {
        submitAnswer();
      }
    });
